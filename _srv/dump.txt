/**
 * servers_ext.php — "Servers / Installations" admin tab.
 * Shows the telemetry each licensed plugin reports (license_telemetry) joined to the license.
 * Licenses are GROUPED BY PHYSICAL SERVER (bound_hostname + bound_ip): a server that runs
 * more than one product (e.g. Safety Monitor + MailShield) shows as ONE card listing all its
 * products, instead of a duplicate card per license. Per-license remote actions (update /
 * restart / revoke / edit) stay available inside the card.
 * Reuses panel helpers: db(), h(), csrf_token(), require_admin(), require_csrf(), redirect(), layout_start/end().

if (!function_exists('db')) return;

function srv_route(string $path): bool {
    $m = $_SERVER['REQUEST_METHOD'] ?? 'GET';
    if ($path === '/servers' && $m === 'GET')          { srv_page(); return true; }
    if ($path === '/servers/cmd' && $m === 'POST')     { srv_cmd();  return true; }
    if ($path === '/servers/cmd-status' && $m === 'GET') { srv_cmd_status(); return true; }
    return false;
}

/** Live status of a queued command (for the push progress bar). */
function srv_cmd_status(): void {
    require_admin();
    header('Content-Type: application/json');
    $id = (int)($_GET['id'] ?? 0);
    if ($id <= 0) { echo json_encode(['ok' => false]); exit; }
    $r = null;
    if ($q = db()->prepare("SELECT command, status, result, sent_at, done_at FROM license_commands WHERE id=? LIMIT 1")) {
        $q->bind_param('i', $id); $q->execute(); $r = $q->get_result()->fetch_assoc();
    }
    if (!$r) { echo json_encode(['ok' => true, 'status' => 'gone']); exit; }   // row removed (e.g. superseded)
    echo json_encode(['ok' => true, 'command' => $r['command'], 'status' => $r['status'], 'result' => (string)$r['result']]);
    exit;
}

function srv_human_uptime($s) { $s = (int)$s; if ($s <= 0) return '—'; $d = floor($s / 86400); $h = floor(($s % 86400) / 3600); return $d > 0 ? ($d . 'd ' . $h . 'h') : ($h . 'h'); }
function srv_ago($dt) { if (!$dt) return '—'; $ts = strtotime($dt); if (!$ts) return h($dt); $s = time() - $ts; if ($s < 60) return $s . 's ago'; if ($s < 3600) return floor($s / 60) . 'm ago'; if ($s < 86400) return floor($s / 3600) . 'h ago'; return floor($s / 86400) . 'd ago'; }

function srv_cmd(): void {
    require_admin(); require_csrf();
    $ajax = ($_POST['ajax'] ?? '') === '1';
    $key = trim((string)($_POST['license_key'] ?? ''));
    $cmd = (string)($_POST['cmd'] ?? '');
    $id = 0;
    if ($key !== '' && in_array($cmd, ['update', 'restart', 'revoke', 'reset', 'disable', 'enable'], true)) {
        $args = $cmd === 'update' ? preg_replace('/[^0-9.]/', '', (string)($_POST['args'] ?? '')) : '';
        // supersede any earlier pending command of the same type (so a newer push wins; no pile-up)
        if ($c = db()->prepare("DELETE FROM license_commands WHERE license_key=? AND command=? AND status='pending'")) {
            $c->bind_param('ss', $key, $cmd); $c->execute();
        }
        if ($st = db()->prepare("INSERT INTO license_commands (license_key, command, args, status) VALUES (?,?,?, 'pending')")) {
            $st->bind_param('sss', $key, $cmd, $args); $st->execute(); $id = (int)db()->insert_id;
        }
        if ($cmd === 'revoke') {
            if ($u = db()->prepare("UPDATE licenses SET status='suspended', updated_at=NOW() WHERE license_key=?")) { $u->bind_param('s', $key); $u->execute(); }
            if ($d = db()->prepare("DELETE FROM license_telemetry WHERE license_key=?")) { $d->bind_param('s', $key); $d->execute(); }
        }
        // RESET: admin investigated a flagged tamper → bring the license back to active + clear the TAMPER notes.
        // The queued 'reset' command tells the plugin to clear its sticky tamper marker and resume.
        if ($cmd === 'reset') {
            if ($u = db()->prepare("UPDATE licenses SET status='active', notes=TRIM(CONCAT_WS(' ', COALESCE(NULLIF(notes,''),''), ?)), updated_at=NOW() WHERE license_key=?")) {
                $note = '[reset ' . date('Y-m-d H:i') . ']'; $u->bind_param('ss', $note, $key); $u->execute();
            }
        }
        if (function_exists('log_event')) log_event($key, $cmd === 'revoke' ? 'revoke' : 'queue_command', 'ok', $cmd . ($args !== '' ? ' ' . $args : ''), []);
    }
    if ($ajax) { header('Content-Type: application/json'); echo json_encode(['ok' => $id > 0, 'id' => $id, 'cmd' => $cmd]); exit; }
    redirect('/servers');
}

/** Group key = the physical server. Bound host+ip is shared by every product on a box;
 *  unbound licenses fall back to their own key so each still gets a card. */
function srv_group_key(array $r): string {
    $host = strtolower(trim((string)($r['bound_hostname'] ?: $r['hostname'] ?: '')));
    $ip   = strtolower(trim((string)($r['bound_ip'] ?: $r['server_ip'] ?: '')));
    if ($host !== '' || $ip !== '') return 'hi:' . $host . '|' . $ip;
    $mac = strtolower(trim((string)($r['mac'] ?? '')));
    if ($mac !== '') return 'mac:' . $mac;
    return 'lk:' . (string)$r['license_key'];
}

/** Available smpro release versions (newest first) for the per-server "push version" picker. */
function srv_versions(): array {
    $out = [];
    foreach (glob('/opt/aitech-license-server/releases/smpro-*.tar.gz') ?: [] as $f)
        if (preg_match('/smpro-([0-9][0-9.]*)\.tar\.gz$/', basename($f), $m)) $out[$m[1]] = 1;
    $out = array_keys($out);
    usort($out, fn($a, $b) => version_compare($b, $a));
    return $out;
}

<?php
/**
 * telemetry_ext.php — Phase A of the Servers/Installations tab.
 * Additive only: stores the latest telemetry snapshot a plugin reports on /api/check,
 * and delivers any pending pull-based command (update/restart/revoke) back to the plugin.
 * Reuses the app's db() (mysqli). Never touches licensing/signing logic.
 *
 * Wire-up in public/index.php:
 *   require_once __DIR__ . '/telemetry_ext.php';   // near the other _ext requires
 * and in api_activate_or_check(), right before the final json_response of the OK path:
 *   if (function_exists('smlic_telemetry_ingest')) smlic_telemetry_ingest($licenseKey, $data, $serverIp, $hostname);
 *   $cmd = function_exists('smlic_command_pop') ? smlic_command_pop($licenseKey) : null;
 *   $resp = ['ok'=>true,'status'=>'active','license'=>$signed]; if ($cmd) $resp['command'] = $cmd;
 *   json_response($resp);
 */

if (!function_exists('db')) return;   // only meaningful inside the app

/** Idempotent schema creation — runs once (guarded by a flag file), reuses the app db(). */
function smlic_telemetry_migrate(): void {
    $flag = __DIR__ . '/../storage/.telemetry_migrated';
    if (is_file($flag)) return;
    try {
        db()->query("CREATE TABLE IF NOT EXISTS license_telemetry (
            license_key VARCHAR(90) NOT NULL PRIMARY KEY, hostname VARCHAR(255) DEFAULT NULL,
            server_ip VARCHAR(45) DEFAULT NULL, mac VARCHAR(64) DEFAULT NULL, os VARCHAR(120) DEFAULT NULL,
            panel VARCHAR(80) DEFAULT NULL, cpu_cores INT DEFAULT NULL, ram_mb INT DEFAULT NULL,
            uptime_sec BIGINT DEFAULT NULL, load1 DECIMAL(7,2) DEFAULT NULL, load15 DECIMAL(7,2) DEFAULT NULL,
            version VARCHAR(20) DEFAULT NULL, reported_at DATETIME DEFAULT NULL, INDEX idx_reported_at (reported_at)
          ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4");
        db()->query("CREATE TABLE IF NOT EXISTS license_commands (
            id BIGINT AUTO_INCREMENT PRIMARY KEY, license_key VARCHAR(90) NOT NULL,
            command ENUM('update','restart','revoke') NOT NULL, args VARCHAR(190) DEFAULT NULL,
            status ENUM('pending','sent','done','failed') NOT NULL DEFAULT 'pending', result TEXT DEFAULT NULL,
            created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP, sent_at DATETIME DEFAULT NULL, done_at DATETIME DEFAULT NULL,
            INDEX idx_key_status (license_key, status), INDEX idx_created_at (created_at)
          ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4");
        @file_put_contents($flag, date('c'));
    } catch (\Throwable $e) { /* will retry next request until the flag is written */ }
}
smlic_telemetry_migrate();

/** UPSERT the latest telemetry snapshot for a license. All fields optional + bounded. */
function smlic_telemetry_ingest(string $key, array $data, string $serverIp, string $hostname): void {
    if ($key === '') return;
    $t = ($data['telemetry'] ?? null);
    if (!is_array($t)) $t = $data;   // accept flat or nested under "telemetry"
    $s   = fn($v, $n) => ($v === null || $v === '') ? null : substr((string)$v, 0, $n);
    $i   = fn($v) => ($v === null || $v === '') ? null : (int)$v;
    $f2  = fn($v) => ($v === null || $v === '') ? null : round((float)$v, 2);
    $mac = $s($t['mac'] ?? null, 64);
    $os  = $s($t['os'] ?? null, 120);
    $pan = $s($t['panel'] ?? null, 80);
    $cor = $i($t['cpu_cores'] ?? null);
    $ram = $i($t['ram_mb'] ?? null);
    $up  = $i($t['uptime_sec'] ?? null);
    $l1  = $f2($t['load1'] ?? null);
    $l15 = $f2($t['load15'] ?? null);
    $ver = $s($t['version'] ?? ($data['plugin_version'] ?? null), 20);
    $host = $s($hostname, 255);
    $ip   = $s($serverIp, 45);
    $sql = "INSERT INTO license_telemetry
              (license_key, hostname, server_ip, mac, os, panel, cpu_cores, ram_mb, uptime_sec, load1, load15, version, reported_at)
            VALUES (?,?,?,?,?,?,?,?,?,?,?,?, NOW())
            ON DUPLICATE KEY UPDATE
              hostname=VALUES(hostname), server_ip=VALUES(server_ip), mac=COALESCE(VALUES(mac),mac),
              os=COALESCE(VALUES(os),os), panel=COALESCE(VALUES(panel),panel),
              cpu_cores=COALESCE(VALUES(cpu_cores),cpu_cores), ram_mb=COALESCE(VALUES(ram_mb),ram_mb),
              uptime_sec=VALUES(uptime_sec), load1=VALUES(load1), load15=VALUES(load15),
              version=COALESCE(VALUES(version),version), reported_at=NOW()";
    try {
        $st = db()->prepare($sql);
        if (!$st) return;
        $st->bind_param('ssssssiiidds', $key, $host, $ip, $mac, $os, $pan, $cor, $ram, $up, $l1, $l15, $ver);
        $st->execute();
    } catch (\Throwable $e) { /* telemetry is best-effort — never break a license check */ }
}

/** Duplicate-binding guard: is another license already bound to this exact IP + hostname?
 *  Returns the other license_key (string) if taken, or null. Used to refuse a 2nd serial on the same server. */
function smlic_server_taken($key, $ip, $host, $fp = null) {
    $ip = trim((string)$ip); $host = trim((string)$host); $fp = trim((string)$fp);
    if ($fp === '') return null; // product-aware: dedup by fingerprint
    try {
        $st = db()->prepare("SELECT license_key FROM licenses WHERE license_key <> ? AND bound_fingerprint = ? AND status <> 'expired' LIMIT 1");
        if (!$st) return null;
        $st->bind_param('ss', $key, $fp); $st->execute();
        $r = $st->get_result()->fetch_assoc();
        return $r ? (string)$r['license_key'] : null;
    } catch (\Throwable $e) { return null; }
}

/** Pop the oldest pending command for a license, mark it 'sent', return it (or null). */
function smlic_command_pop(string $key): ?array {
    if ($key === '') return null;
    try {
        $st = db()->prepare("SELECT id, command, args FROM license_commands WHERE license_key=? AND status='pending' ORDER BY id ASC LIMIT 1");
        if (!$st) return null;
        $st->bind_param('s', $key); $st->execute();
        $r = $st->get_result()->fetch_assoc();
        if (!$r) return null;
        $up = db()->prepare("UPDATE license_commands SET status='sent', sent_at=NOW() WHERE id=?");
        $id = (int)$r['id']; $up->bind_param('i', $id); $up->execute();
        return ['id' => (int)$r['id'], 'command' => (string)$r['command'], 'args' => (string)($r['args'] ?? '')];
    } catch (\Throwable $e) { return null; }
}

/** Mark a command done/failed (called from a future /api/command-ack; defined now for completeness). */
function smlic_command_ack(int $id, string $status, string $result = ''): void {
    if (!in_array($status, ['done', 'failed'], true)) return;
    try {
        $st = db()->prepare("UPDATE license_commands SET status=?, result=?, done_at=NOW() WHERE id=?");
        $st->bind_param('ssi', $status, $result, $id); $st->execute();
    } catch (\Throwable $e) {}
}
